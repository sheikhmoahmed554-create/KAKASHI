//  KAKASHI LEVELS V2 — XAUUSD
//
//  Everything here was chosen blind. For each month, the configuration was
//  searched on the months already closed, frozen, and then traded through a
//  month it had never seen. Five such months, stitched together:
//
//      776 trades   53.99%   +8,058 points   PF 1.225   worst drawdown 1,178
//      four of five months positive, no month-on-month decay
//
//  For contrast, AI 963 measured on the same candles: 7,687 trades, 52.02%,
//  +837 points, drawdown 10,839. Nine times the drawdown for a tenth of the
//  profit, from ten times the trades.
//
//  The three sources, and why each is here:
//
//    Falling trendline, 15m, bounce.  +23.74 per trade over 125. Every one of
//      the five folds chose 15m and chose the bounce, independently, without
//      seeing what came after — the most stable selection in the study.
//
//    Fair value gaps, 60m, fade the break.  +6.10 per trade over 408. Every
//      fold chose to FADE the break rather than follow it: a gap giving way is
//      exhaustion, not continuation. Positive in all five unseen months.
//
//    Rising trendline, 15m, break.  +10.70 per trade over 243, and the weakest
//      claim of the three: its chosen timeframe and reading changed in every
//      fold, so its profit came from a different rule each month. It is kept
//      because the portfolio measures better with it than without — adjusted
//      t 1.77 against 1.45, and a smaller drawdown — but of the three this is
//      the one to switch off first.
//
//  WHAT IS STILL MISSING: significance. The adjusted t is 1.77 against the 1.96
//  that would put the odds of chance under 5%. This is the closest the work
//  came, and it did not arrive. Direction is not the explanation — longs earn
//  +10.62 over a blind long and shorts +9.24 over a blind short, so the falling
//  market is not doing the work — but seven months of one instrument is a thin
//  sample, and absorption already showed how a strong in-sample source can be
//  negative out of it.
//
//  Trade it small, and judge it on the next hundred trades rather than on this
//  header.
//

// ═════════════════════════════════════════════════════════════════════════════
//  The event detector, shared by all three.
//
//  A level price loitered at for forty bars is not forty rejections, and a
//  candle that merely happens to be near a level was never a test of it.
//  Counting per candle instead of per visit is what made an earlier study
//  produce 50,576 signals from round numbers across 192,081 candles.
//
//  Arm only once price is genuinely away, fire once when it arrives and reacts,
//  then lock until it leaves. Pine keeps these var's per call site, so each
//  source carries its own state with no bookkeeping here.
// ═════════════════════════════════════════════════════════════════════════════
f_events(float lvl, float atrNow) =>
    var float armedLevel = na
    var float lockedLevel = na
    var bool  approached = false
    int rejDir = 0
    int brkDir = 0

    if not na(lvl) and not na(atrNow) and atrNow > 0
        if na(armedLevel) or math.abs(lvl - armedLevel) > math.max(0.05, atrNow * 0.5)
            armedLevel := lvl
            approached := false
            lockedLevel := na

        tol = atrNow * tolAtr
        if not na(lockedLevel) and math.abs(close - lockedLevel) > atrNow * resetAtr
            lockedLevel := na
        if math.abs(close - lvl) >= atrNow * approachAtr
            approached := true

        if approached and na(lockedLevel) and not na(close[1])
            fromAbove = close[1] > lvl
            reached = fromAbove ? low <= lvl + tol : high >= lvl - tol
            if reached
                if fromAbove
                    if close > lvl + tol * 0.5
                        rejDir := 1
                    else if close < lvl - atrNow * breakAtr
                        brkDir := -1
                else
                    if close < lvl - tol * 0.5
                        rejDir := -1
                    else if close > lvl + atrNow * breakAtr
                        brkDir := 1
                if rejDir != 0 or brkDir != 0
                    lockedLevel := lvl
                    approached := false
    [rejDir, brkDir]

// ═════════════════════════════════════════════════════════════════════════════
//  Trendlines. Two confirmed pivots sloping the right way, joined and extended.
//  A pivot is only known tlPiv bars after it happened, and the ray is rejected
//  unless nothing between the anchors pierced it — a line price already broke
//  was never a line.
// ═════════════════════════════════════════════════════════════════════════════
f_trendline(bool rising) =>
    float pv = rising ? ta.pivotlow(low, tlPiv, tlPiv) : ta.pivothigh(high, tlPiv, tlPiv)
    var float[] pxs = array.new_float()
    var int[]   pbs = array.new_int()
    var float ax = na
    var float ay = na
    var float slope = na
    var int   bornAt = na

    if not na(pv)
        int pb = bar_index - tlPiv
        float pp = pv
        int n = array.size(pxs)
        bool found = false
        // Pine's for-loop counts DOWN when `to` is below `from`, so with an
        // empty pool `for step = 1 to 0` still runs once and reads index -1.
        // The first pivot of the chart hits exactly that.
        if n > 0
            for step = 1 to n
                if not found
                    int k = n - step
                    float qp = array.get(pxs, k)
                    int   qb = array.get(pbs, k)
                    int   span = pb - qb
                    if span > tlMaxSpan
                        break
                    bool slopeOK = rising ? pp > qp : pp < qp
                    if span >= tlMinSpan and slopeOK
                        float sl2 = (pp - qp) / span
                        bool clean = true
                        for j = 0 to span
                            int back = bar_index - (qb + j)
                            if back >= 0
                                float ry = qp + sl2 * j
                                bool pierced = rising ? low[back] < ry - nz(atrv) * tlPierce
                                                      : high[back] > ry + nz(atrv) * tlPierce
                                if pierced
                                    clean := false
                                    break
                        if clean
                            ax := pb
                            ay := pp
                            slope := sl2
                            bornAt := bar_index
                            found := true
        array.push(pxs, pp)
        array.push(pbs, pb)
        if array.size(pxs) > 60
            array.shift(pxs)
            array.shift(pbs)

    float y = na
    if not na(ax)
        y := ay + slope * (bar_index - ax)
        if bar_index - bornAt > tlMaxProj
            ax := na
            y := na
        else if rising ? close < y - nz(atrv) * tlBreakA : close > y + nz(atrv) * tlBreakA
            ax := na
    // Lagged inside the function rather than by indexing the call, so the value
    // reaching the chart belongs to a candle that has closed.
    y[1]

// ═════════════════════════════════════════════════════════════════════════════
//  Fair value gap: three candles where the middle one moved far enough that the
//  outer two do not overlap. The untouched middle is what price comes back for.
//  Traded as a fade of the break, which is what every fold independently chose.
// ═════════════════════════════════════════════════════════════════════════════
f_fvg() =>
    var float lvl = na
    var int   born = na
    float a = nz(atrv)
    float upGap = low - high[2]
    float dnGap = low[2] - high
    if a > 0 and upGap >= a * fvgMin
        lvl := (low + high[2]) / 2
        born := bar_index
    else if a > 0 and dnGap >= a * fvgMin
        lvl := (high + low[2]) / 2
        born := bar_index
    if not na(born) and bar_index - born > fvgAge
        lvl := na
    lvl[1]

// ─────────────────────────────────────────────────────────────────────────────
//  One virtual position per source: target, stop, and a time stop. The time
//  stop matters — these were measured with a hold limit, and a trade left open
//  past it is not the trade that was measured.
// ─────────────────────────────────────────────────────────────────────────────
barMinutes = timeframe.in_seconds(timeframe.period) / 60

f_manage(int sig, float tp, float sl, int holdMin, bool enabled) =>
    var int   dir = 0
    var float entry = na
    var float tpPx = na
    var float slPx = na
    var int   openBar = na
    int  result = 0
    float pts = 0.0
    bool opened = false

    if dir != 0 and bar_index > openBar
        bool hitTP = dir == 1 ? high >= tpPx : low <= tpPx
        bool hitSL = dir == 1 ? low <= slPx : high >= slPx
        if hitTP and hitSL
            result := -1              // both in one candle: read it as the stop
            pts := -sl
        else if hitTP
            result := 1
            pts := tp
        else if hitSL
            result := -1
            pts := -sl
        else if (bar_index - openBar) * barMinutes >= holdMin
            result := 2
            pts := (close - entry) * dir / pointUnit
        if result != 0
            dir := 0
            entry := na
            tpPx := na
            slPx := na

    if enabled and dir == 0 and sig != 0 and barstate.isconfirmed
        dir := sig
        entry := close
        tpPx := close + sig * tp * pointUnit
        slPx := close - sig * sl * pointUnit
        openBar := bar_index
        opened := true
    [dir, entry, tpPx, slPx, result, pts, opened]

// A trade can close and the next one open on the same candle. Reading the entry
// off `dir != 0 and dir[1] == 0` would miss exactly that case and leave a live
// position with no label, no lines and no alert, so the flag is raised where the
// entry actually happens.
[dnDir, dnEntry, dnTPpx, dnSLpx, dnRes, dnPts, enteredDN] = f_manage(dnSig, dnTP, dnSL, dnHold, useDN)
[fvDir, fvEntry, fvTPpx, fvSLpx, fvRes, fvPts, enteredFV] = f_manage(fvgSig, fvgTP, fvgSL, fvgHold, useFVG)
[upDir, upEntry, upTPpx, upSLpx, upRes, upPts, enteredUP] = f_manage(upSig, upTP, upSL, upHold, useUP)

// ─────────────────────────────────────────────────────────────────────────────

// One drawing set per source. Pine keeps these var's per call site, so each
// source owns its own lines without any index bookkeeping.
// The close is handled before the open, because both can land on one candle and
// clearing afterwards would erase the drawing of the trade that just started.
f_draw(bool opened, int dir, float entry, float tpPx, float slPx, int res, string tag) =>
    var line lnE = na
    var line lnT = na
    var line lnS = na
    var label lbE = na
    var label lbT = na
    var label lbS = na

    if res != 0
        if not na(lnE)
            line.delete(lnE)
            lnE := na
        if not na(lnT)
            line.delete(lnT)
            lnT := na
        if not na(lnS)
            line.delete(lnS)
            lnS := na
        if not na(lbE)
            label.delete(lbE)
            lbE := na
        if not na(lbT)
            label.delete(lbT)
            lbT := na
        if not na(lbS)
            label.delete(lbS)
            lbS := na

    if opened and showTrades
        int x2 = bar_index + lineBars
        lnE := line.new(bar_index, entry, x2, entry, xloc = xloc.bar_index, color = cEntry, width = 1, style = line.style_solid)
        lnT := line.new(bar_index, tpPx,  x2, tpPx,  xloc = xloc.bar_index, color = cTP,   width = 1, style = line.style_dashed)
        lnS := line.new(bar_index, slPx,  x2, slPx,  xloc = xloc.bar_index, color = cSell, width = 1, style = line.style_dashed)
        if showTags
            string side = dir == 1 ? "شراء" : "بيع"
            lbE := label.new(x2, entry, "دخول " + side + " · " + tag, xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_left, color = cEntry, textcolor = color.black, size = size.tiny)
            lbT := label.new(x2, tpPx,  "هدف · " + tag, xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_left, color = cTP,   textcolor = color.white, size = size.tiny)
            lbS := label.new(x2, slPx,  "وقف · " + tag, xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_left, color = cSell, textcolor = color.white, size = size.tiny)
            label.set_tooltip(lbE, tag + " — " + side + " عند " + str.tostring(entry, format.mintick))
    opened

f_draw(enteredDN, dnDir, dnEntry, dnTPpx, dnSLpx, dnRes, "ترند هابط")
